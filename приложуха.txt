<script>
const compliments = [
    "Звёздочка, ты — мой лучик света! ☀️",
    "Звёздочка, твоя улыбка делает мир ярче! 😊",
    "Звёздочка, ты самая лучшая на свете! ❤️",
    "Звёздочка, с тобой каждый день — праздник! 🎉",
    "Звёздочка, ты невероятная! ✨",
    "Звёздочка, я так рад, что ты у меня есть! 💕",
    "Звёздочка, ты — моё счастье! 🦋",
    "Звёздочка, ты — волшебство! 🌟",
    "Звёздочка, обожаю тебя! 💖",
    "Звёздочка, ты делаешь меня счастливым! 🥰",
    "Звёздочка, твоя доброта безгранична! 🌺"
];

const canvas = document.getElementById('heartsCanvas');
const ctx = canvas.getContext('2d');
let W, H;
function resize() { W = canvas.width = innerWidth; H = canvas.height = innerHeight; }
resize();
addEventListener('resize', resize);

const symbols = ['❤️','💕','💖','💗','💝'];

class FallingHeart {
    constructor() { this.reset(true); }
    reset(init) {
        this.x = Math.random() * W;
        this.y = init ? Math.random() * H : -50;
        this.size = 20 + Math.random() * 25;
        this.speed = 1.5 + Math.random() * 3.5;
        this.sway = Math.random() * Math.PI * 2;
        this.symbol = symbols[Math.floor(Math.random() * symbols.length)];
        this.opacity = 0.5 + Math.random() * 0.5;
    }
    update() {
        this.y += this.speed;
        this.sway += 0.03;
        this.x += Math.sin(this.sway) * 1;
        if (this.y > H + 50) this.reset();
    }
    draw() {
        ctx.save();
        ctx.globalAlpha = this.opacity;
        ctx.font = this.size + 'px Arial';
        ctx.textAlign = 'center';
        ctx.fillText(this.symbol, this.x, this.y);
        ctx.restore();
    }
}

const hearts = [];
for (let i = 0; i < 25; i++) hearts.push(new FallingHeart());

function animate() {
    ctx.clearRect(0, 0, W, H);
    hearts.forEach(h => { h.update(); h.draw(); });
    requestAnimationFrame(animate);
}
animate();

function explosion(x, y) {
    const syms = ['❤️','💕','💖','✨','⭐','💗'];
    for (let i = 0; i < 25; i++) {
        const el = document.createElement('div');
        el.className = 'explosion-heart';
        el.textContent = syms[Math.floor(Math.random() * syms.length)];
        el.style.left = x + 'px';
        el.style.top = y + 'px';
        const angle = (Math.PI * 2 * i) / 25;
        const dist = 80 + Math.random() * 200;
        el.style.setProperty('--tx', Math.cos(angle) * dist + 'px');
        el.style.setProperty('--ty', Math.sin(angle) * dist + 'px');
        document.body.appendChild(el);
        setTimeout(() => el.remove(), 1500);
    }
}

const complimentEl = document.getElementById('compliment');
let last = '';

function showCompliment(x, y) {
    let c;
    do { c = compliments[Math.floor(Math.random() * compliments.length)]; }
    while (c === last && compliments.length > 1);
    last = c;
    complimentEl.textContent = c;
    
    const heart = document.getElementById('bigHeart');
    const rect = heart.getBoundingClientRect();
    explosion(rect.left + rect.width/2, rect.top + rect.height/2);
}

document.getElementById('bigHeart').addEventListener('click', showCompliment);
document.getElementById('btnMore').addEventListener('click', showCompliment);

setTimeout(showCompliment, 1000);
